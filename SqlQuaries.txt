CREATE TABLE students (
    student_id INT IDENTITY(1,1) PRIMARY KEY, 
    full_name VARCHAR(255) NOT NULL, 
    email VARCHAR(100) UNIQUE,
    phone VARCHAR(15) CHECK (LEN(phone) >= 10), 
    grade VARCHAR(20) NOT NULL,
    address VARCHAR(255),
    dob DATE,
    guardian_name VARCHAR(255),
    guardian_phone VARCHAR(15),
    status VARCHAR(20) DEFAULT 'active' CHECK (status IN ('active', 'inactive')),
    username VARCHAR(50) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    created_at DATETIME DEFAULT GETDATE()
);

CREATE VIEW vw_IncomePrediction AS
SELECT 
    s.grade,
    COUNT(e.student_id) AS Total_Students,
    SUM(c.fee) AS Expected_Monthly_Income,
    SUM(CASE WHEN e.payment_status = 'pending' THEN c.fee ELSE 0 END) AS Outstanding_Amount
FROM enrollments e
JOIN students s ON e.student_id = s.student_id
JOIN classes c ON e.class_id = c.class_id
GROUP BY s.grade;
GO

CREATE OR ALTER PROCEDURE sp_AnalyzeStudentRisk
AS
BEGIN
    SELECT 
        s.full_name,
        s.grade,
      
        COUNT(p.payment_id) AS Payments_Made,
        -- Data Mining Classification Logic
        CASE 
            WHEN COUNT(p.payment_id) >= 10 THEN 'Safe'
            WHEN COUNT(p.payment_id) BETWEEN 5 AND 9 THEN 'Warning'
            ELSE 'Critical (High Risk of Churn)'
        END AS Risk_Status
    FROM students s
    LEFT JOIN enrollments e ON s.student_id = e.student_id
    LEFT JOIN payments p ON e.enroll_id = p.enroll_id
    GROUP BY s.full_name, s.grade;
END;
GO
